cmake_minimum_required(VERSION 3.16)
project(rms_audio_host VERSION 0.1.0 LANGUAGES CXX)
option(RMS_CORE_ONLY "Build tests without JACK or FluidSynth development files" OFF)
option(RMS_SANITIZE "Enable address/undefined sanitizers for native tests" OFF)
set(RMS_VST3_SDK "" CACHE PATH "Explicit pinned VST3 SDK source directory (optional)")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/nlohmann/json.hpp" RMS_JSON_SHA256)
if(NOT RMS_JSON_SHA256 STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
    message(FATAL_ERROR "Vendored nlohmann/json 3.11.3 digest mismatch")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(RMS_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(rms-core STATIC core.cpp)
target_include_directories(rms-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
include(CTest)
if(BUILD_TESTING)
    add_executable(rms-core-tests core_tests.cpp)
    target_link_libraries(rms-core-tests PRIVATE rms-core)
    add_test(NAME scheduler-core COMMAND rms-core-tests)
endif()
# Core scheduling tests have zero external dependencies. Protocol tests use the
# same optional system OpenSSL dependency as the live host, not an audio SDK.
find_package(OpenSSL QUIET COMPONENTS Crypto)
if(OpenSSL_FOUND)
    find_package(Threads REQUIRED)
    add_library(rms-protocol STATIC protocol.cpp)
    target_include_directories(rms-protocol SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../vendor")
    target_link_libraries(rms-protocol PUBLIC rms-core OpenSSL::Crypto Threads::Threads)
    add_library(rms-vst3 STATIC vst3.cpp)
    target_link_libraries(rms-vst3 PUBLIC rms-protocol)
    if(RMS_VST3_SDK)
        if(CMAKE_VERSION VERSION_LESS 3.25)
            message(FATAL_ERROR "SDK builds require CMake >=3.25")
        endif()
        find_package(Git REQUIRED)
        foreach(pair IN ITEMS ".|3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96"
                "base|fcf9da0bd27a16f7f03773a3a39822f28f5c8477"
                "cmake|054c9143cbb8d47fc4694e473f2ee3b4d951a8f5"
                "pluginterfaces|4f547e8e102b47de4a8b8aaf343c73b700786372"
                "public.sdk|586dc5e6c8012c3e4b01c79389375cbe96bdb1da")
            string(REPLACE "|" ";" revision "${pair}")
            list(GET revision 0 directory)
            list(GET revision 1 expected)
            execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${RMS_VST3_SDK}/${directory}" rev-parse HEAD
                OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE code)
            if(NOT code EQUAL 0 OR NOT actual STREQUAL expected)
                message(FATAL_ERROR "SDK revision mismatch in ${directory}: expected ${expected}")
            endif()
            execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${RMS_VST3_SDK}/${directory}" diff --quiet HEAD
                RESULT_VARIABLE dirty)
            if(NOT dirty EQUAL 0)
                message(FATAL_ERROR "SDK tracked sources modified in ${directory}")
            endif()
        endforeach()
        set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
        set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
        set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
        set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
        add_subdirectory("${RMS_VST3_SDK}" "${CMAKE_BINARY_DIR}/vst3-sdk" EXCLUDE_FROM_ALL)
        target_sources(sdk_hosting PRIVATE "${RMS_VST3_SDK}/public.sdk/source/vst/hosting/plugprovider.cpp")
        foreach(target IN ITEMS base pluginterfaces sdk_common sdk sdk_hosting)
            target_compile_options(${target} PRIVATE -Wno-error)
        endforeach()
        target_compile_definitions(rms-vst3 PRIVATE RMS_VST3=1)
        target_include_directories(rms-vst3 SYSTEM PRIVATE "${RMS_VST3_SDK}")
        target_sources(rms-vst3 PRIVATE "${RMS_VST3_SDK}/public.sdk/source/vst/hosting/module_linux.cpp")
        target_link_libraries(rms-vst3 PRIVATE sdk_hosting sdk dl)
        find_package(X11 QUIET)
        if(X11_FOUND)
            target_compile_definitions(rms-vst3 PRIVATE RMS_X11=1)
            target_link_libraries(rms-vst3 PRIVATE X11::X11)
        endif()
        if(BUILD_TESTING)
            smtg_add_vst3plugin(rms-test-instrument test_instrument.cpp)
            target_sources(rms-test-instrument PRIVATE "${RMS_VST3_SDK}/public.sdk/source/main/linuxmain.cpp")
            target_link_libraries(rms-test-instrument PRIVATE sdk)
            if(X11_FOUND)
                target_compile_definitions(rms-test-instrument PRIVATE RMS_X11=1)
                target_link_libraries(rms-test-instrument PRIVATE X11::X11)
            endif()
            target_include_directories(rms-test-instrument SYSTEM PRIVATE "${RMS_VST3_SDK}")
            add_executable(rms-vst3-tests vst3_tests.cpp)
            target_link_libraries(rms-vst3-tests PRIVATE rms-vst3)
            # Explicit opt-in real sfizz test; never registered in CTest and
            # never downloads a plugin or opens JACK/audio hardware.
            add_executable(rms-sfizz-verify EXCLUDE_FROM_ALL sfizz_verify.cpp)
            target_link_libraries(rms-sfizz-verify PRIVATE rms-vst3)
            if(X11_FOUND)
                target_compile_definitions(rms-vst3-tests PRIVATE RMS_EDITOR_TEST=1)
                target_link_libraries(rms-vst3-tests PRIVATE X11::X11)
            endif()
            add_dependencies(rms-vst3-tests rms-test-instrument)
            add_test(NAME vst3-sdk-abi-offline COMMAND rms-vst3-tests
                "${CMAKE_BINARY_DIR}/VST3/$<CONFIG>/rms-test-instrument.vst3")
            if(X11_FOUND)
                add_test(NAME vst3-editor-x11 COMMAND rms-vst3-tests
                    "${CMAKE_BINARY_DIR}/VST3/$<CONFIG>/rms-test-instrument.vst3" --editor)
                set_tests_properties(vst3-editor-x11 PROPERTIES TIMEOUT 15 SKIP_RETURN_CODE 77)
            endif()
        endif()
    endif()
    if(BUILD_TESTING)
        add_executable(rms-protocol-tests protocol_tests.cpp)
        target_link_libraries(rms-protocol-tests PRIVATE rms-protocol)
        add_test(NAME protocol-validation COMMAND rms-protocol-tests)
    endif()
elseif(NOT RMS_CORE_ONLY)
    message(FATAL_ERROR "Live host requires OpenSSL development headers and libcrypto (SHA256). Use --core-only for scheduler tests.")
else()
    message(STATUS "OpenSSL development files absent: scheduler tests only")
endif()
if(NOT RMS_CORE_ONLY)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(JACK QUIET IMPORTED_TARGET jack)
    if(NOT JACK_FOUND)
        message(FATAL_ERROR "Live host requires JACK development headers/pkg-config metadata. Use --core-only for tests.")
    endif()
    # Explicit overrides support local validation against extracted official
    # distribution headers without installing or changing system packages.
    set(FLUIDSYNTH_INCLUDE_DIR "" CACHE PATH "Directory containing fluidsynth.h")
    set(FLUIDSYNTH_LIBRARY "" CACHE FILEPATH "Absolute path to libfluidsynth.so")
    if(FLUIDSYNTH_INCLUDE_DIR AND FLUIDSYNTH_LIBRARY)
        add_library(rms-fluid UNKNOWN IMPORTED)
        set_target_properties(rms-fluid PROPERTIES IMPORTED_LOCATION "${FLUIDSYNTH_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${FLUIDSYNTH_INCLUDE_DIR}")
        set(RMS_FLUID_TARGET rms-fluid)
    else()
        pkg_check_modules(FLUID QUIET IMPORTED_TARGET fluidsynth>=2.2)
        if(NOT FLUID_FOUND)
            message(FATAL_ERROR "Live host requires libfluidsynth >=2.2 development headers and fluidsynth.pc; runtime-only is insufficient. Use --core-only for tests. No dependencies are downloaded or installed.")
        endif()
        set(RMS_FLUID_TARGET PkgConfig::FLUID)
    endif()
    add_executable(rms-audio-host main.cpp live.cpp)
    target_link_libraries(rms-audio-host PRIVATE rms-protocol rms-vst3 PkgConfig::JACK "${RMS_FLUID_TARGET}")
    if(BUILD_TESTING)
        # A link-time JACK double exercises absent server/ports without opening
        # any audio device. It is never linked into the production helper.
        add_executable(rms-live-failure-tests live.cpp live_failure_tests.cpp)
        target_include_directories(rms-live-failure-tests PRIVATE ${JACK_INCLUDE_DIRS})
        target_link_libraries(rms-live-failure-tests PRIVATE rms-protocol rms-vst3 "${RMS_FLUID_TARGET}")
        add_test(NAME live-failures-no-audio COMMAND rms-live-failure-tests)
        find_package(Python3 QUIET COMPONENTS Interpreter)
        if(Python3_FOUND)
            add_test(NAME helper-wire-process
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/smoke.py"
                    --binary "$<TARGET_FILE:rms-audio-host>")
        endif()
    endif()
endif()
