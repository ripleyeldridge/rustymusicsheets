cmake_minimum_required(VERSION 3.16)
project(rms_audio_host VERSION 0.1.0 LANGUAGES CXX)
option(RMS_CORE_ONLY "Build tests without JACK or FluidSynth development files" OFF)
option(RMS_SANITIZE "Enable address/undefined sanitizers for native tests" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/nlohmann/json.hpp" RMS_JSON_SHA256)
if(NOT RMS_JSON_SHA256 STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
    message(FATAL_ERROR "Vendored nlohmann/json 3.11.3 digest mismatch")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(RMS_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(rms-core STATIC core.cpp)
target_include_directories(rms-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
include(CTest)
if(BUILD_TESTING)
    add_executable(rms-core-tests core_tests.cpp)
    target_link_libraries(rms-core-tests PRIVATE rms-core)
    add_test(NAME scheduler-core COMMAND rms-core-tests)
endif()
# Core scheduling tests have zero external dependencies. Protocol tests use the
# same optional system OpenSSL dependency as the live host, not an audio SDK.
find_package(OpenSSL QUIET COMPONENTS Crypto)
if(OpenSSL_FOUND)
    add_library(rms-protocol STATIC protocol.cpp)
    target_include_directories(rms-protocol SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../vendor")
    target_link_libraries(rms-protocol PUBLIC rms-core OpenSSL::Crypto)
    if(BUILD_TESTING)
        add_executable(rms-protocol-tests protocol_tests.cpp)
        target_link_libraries(rms-protocol-tests PRIVATE rms-protocol)
        add_test(NAME protocol-validation COMMAND rms-protocol-tests)
    endif()
elseif(NOT RMS_CORE_ONLY)
    message(FATAL_ERROR "Live host requires OpenSSL development headers and libcrypto (SHA256). Use --core-only for scheduler tests.")
else()
    message(STATUS "OpenSSL development files absent: scheduler tests only")
endif()
if(NOT RMS_CORE_ONLY)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(JACK QUIET IMPORTED_TARGET jack)
    if(NOT JACK_FOUND)
        message(FATAL_ERROR "Live host requires JACK development headers/pkg-config metadata. Use --core-only for tests.")
    endif()
    # Explicit overrides support local validation against extracted official
    # distribution headers without installing or changing system packages.
    set(FLUIDSYNTH_INCLUDE_DIR "" CACHE PATH "Directory containing fluidsynth.h")
    set(FLUIDSYNTH_LIBRARY "" CACHE FILEPATH "Absolute path to libfluidsynth.so")
    if(FLUIDSYNTH_INCLUDE_DIR AND FLUIDSYNTH_LIBRARY)
        add_library(rms-fluid UNKNOWN IMPORTED)
        set_target_properties(rms-fluid PROPERTIES IMPORTED_LOCATION "${FLUIDSYNTH_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${FLUIDSYNTH_INCLUDE_DIR}")
        set(RMS_FLUID_TARGET rms-fluid)
    else()
        pkg_check_modules(FLUID QUIET IMPORTED_TARGET fluidsynth>=2.2)
        if(NOT FLUID_FOUND)
            message(FATAL_ERROR "Live host requires libfluidsynth >=2.2 development headers and fluidsynth.pc; runtime-only is insufficient. Use --core-only for tests. No dependencies are downloaded or installed.")
        endif()
        set(RMS_FLUID_TARGET PkgConfig::FLUID)
    endif()
    add_executable(rms-audio-host main.cpp live.cpp)
    target_link_libraries(rms-audio-host PRIVATE rms-protocol PkgConfig::JACK "${RMS_FLUID_TARGET}")
    if(BUILD_TESTING)
        # A link-time JACK double exercises absent server/ports without opening
        # any audio device. It is never linked into the production helper.
        add_executable(rms-live-failure-tests live.cpp live_failure_tests.cpp)
        target_include_directories(rms-live-failure-tests PRIVATE ${JACK_INCLUDE_DIRS})
        target_link_libraries(rms-live-failure-tests PRIVATE rms-protocol "${RMS_FLUID_TARGET}")
        add_test(NAME live-failures-no-audio COMMAND rms-live-failure-tests)
        find_package(Python3 QUIET COMPONENTS Interpreter)
        if(Python3_FOUND)
            add_test(NAME helper-wire-process
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/smoke.py"
                    --binary "$<TARGET_FILE:rms-audio-host>")
        endif()
    endif()
endif()
